① アニメーション — ワークフロー追加コード
Runway ML等で作ったMP4をアップロード後、以下をワークフローに追加してください
→ ワークフローを編集（GitHub Web UI）
- name: Use animation loop if available
  run: |
    if ls animations/*.mp4 2>/dev/null | grep -q .; then
      ANIM=$(ls animations/*.mp4 | shuf -n 1)
      echo "ANIM_FILE=$ANIM" >> $GITHUB_ENV
      echo "USE_ANIM=true" >> $GITHUB_ENV
    else
      echo "USE_ANIM=false" >> $GITHUB_ENV
    fi

# 「Compose final video」ステップの run: の先頭に追加:
    if [ "$USE_ANIM" = "true" ]; then
      ffmpeg -stream_loop -1 -i "$ANIM_FILE" \
        -i audio.mp3 -shortest \
        -vf "scale=1280:720:force_original_aspect_ratio=decrease,pad=1280:720:(ow-iw)/2:(oh-ih)/2" \
        -c:v libx264 -c:a aac -b:a 192k output.mp4
    else
